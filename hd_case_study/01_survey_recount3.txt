> ## ---- 5. helpers: parse SRA sample attributes --------------------------------
> # recount3 stores SRA phenotype in colData()$sra.sample_attributes as
> # "key1;;value1|key2;;value2|...". Parse into a long (sample_idx, key, value) frame.
> parse_attr <- function(x) {
+   parts <- strsplit(ifelse(is.na(x), "", as.character(x)), "\\|")
+   rows <- lapply(seq_along(parts), function(i) {
+     kv <- parts[[i]]
+     kv <- kv[grepl(";;", kv, fixed = TRUE)]
+     if (!length(kv)) return(NULL)
+     sp  <- strsplit(kv, ";;", fixed = TRUE)
+     key <- vapply(sp, function(z) tolower(trimws(z[1])), character(1))
+     val <- vapply(sp, function(z) trimws(paste(z[-1], collapse = ";;")), character(1))
+     data.frame(sample_idx = i, key = key, value = val, stringsAsFactors = FALSE)
+   })
+   do.call(rbind, rows)   # rbind ignores NULLs; returns NULL if all empty
+ }
>
> BLOOD_RX     <- "whole blood|peripheral blood|pbmc|buffy|\\bblood\\b"

> inspect_sra_study <- function(SRP) {
+   LOG("\n== inspecting SRA study ", SRP, " ==")
+   row <- proj[proj$project == SRP & proj$file_source == "sra", , drop = FALSE]
+   if (nrow(row) == 0L) {
+     LOG("  NOT FOUND in recount3 (frozen ~Oct-2019 snapshot). skip."); return(NULL)
+   }
+   rse <- tryCatch(recount3::create_rse(row),
+                   error = function(e) { LOG("  create_rse failed: ", conditionMessage(e)); NULL })
+   if (is.null(rse)) return(NULL)
+   cd <- as.data.frame(SummarizedExperiment::colData(rse))
+   n  <- nrow(cd)
+   attr_col <- intersect(c("sra.sample_attributes", "sample_attributes"), colnames(cd))[1]
+   kv <- if (!is.na(attr_col)) parse_attr(cd[[attr_col]]) else NULL
+
+   pick_field <- function(keys) {
+     if (is.null(kv)) return(NULL)
+     hit <- kv[kv$key %in% keys, , drop = FALSE]
+     if (!nrow(hit)) return(NULL)
+     best <- names(sort(table(hit$key), decreasing = TRUE))[1]  # key covering most samples
+     v  <- rep(NA_character_, n)
+     hh <- hit[hit$key == best, , drop = FALSE]
+     v[hh$sample_idx] <- hh$value
+     list(key = best, value = v)
+   }
+   tissue  <- pick_field(TISSUE_KEYS)
+   disease <- pick_field(DISEASE_KEYS)
+
+   # blood detection: scan tissue field + all attribute values + any sra.* char cols
+   scan_txt <- c(if (!is.null(tissue)) tissue$value,
+                 if (!is.null(kv)) kv$value,
+                 unlist(lapply(cd[grepl("^sra\\.", colnames(cd))], as.character), use.names = FALSE))
+   blood_hits <- any(grepl(BLOOD_RX, scan_txt, ignore.case = TRUE), na.rm = TRUE)
+
+   tissue_tab  <- if (!is.null(tissue))  sort(table(tissue$value),  decreasing = TRUE) else NULL
+   disease_tab <- if (!is.null(disease)) sort(table(disease$value), decreasing = TRUE) else NULL
+
+   LOG("  samples: ", n)
+   LOG("  tissue field : ", if (!is.null(tissue))  tissue$key  else "<none found>")
+   if (!is.null(tissue_tab))
+     LOG("    ", paste(sprintf("%s=%d", names(head(tissue_tab, 8)), head(tissue_tab, 8)), collapse = " | "))
+   LOG("  disease field: ", if (!is.null(disease)) disease$key else "<none found>")
+   if (!is.null(disease_tab))
+     LOG("    ", paste(sprintf("%s=%d", names(head(disease_tab, 8)), head(disease_tab, 8)), collapse = " | "))
+   LOG("  whole-blood mentioned: ", blood_hits)
+
+   per <- data.frame(
+     sample_id = rownames(cd),
+     tissue    = if (!is.null(tissue))  tissue$value  else NA_character_,
+     disease   = if (!is.null(disease)) disease$value else NA_character_,
+     row.names = NULL
+   )
+   write.csv(per, file.path(OUT_DIR, paste0("sra_", SRP, "_summary.csv")), row.names = FALSE)
+
+   data.frame(
+     SRP = SRP, n_samples = n,
+     tissue_field     = if (!is.null(tissue))  tissue$key  else NA_character_,
+     disease_field    = if (!is.null(disease)) disease$key else NA_character_,
+     n_tissue_levels  = if (!is.null(tissue_tab))  length(tissue_tab)  else 0L,
+     n_disease_levels = if (!is.null(disease_tab)) length(disease_tab) else 0L,
+     whole_blood      = blood_hits,
+     top_tissue       = if (!is.null(tissue_tab))  names(tissue_tab)[1]  else NA_character_,
+     top_disease      = if (!is.null(disease_tab)) names(disease_tab)[1] else NA_character_,
+     row.names = NULL
+   )
+ }

> tcga_sample_types <- function(project) {
+   LOG("\n== TCGA ", project, " sample types ==")
+   row <- proj[proj$project == project & proj$file_source == "tcga", , drop = FALSE]
+   if (nrow(row) == 0L) { LOG("  not a TCGA project in recount3. skip."); return(NULL) }
+   rse <- tryCatch(recount3::create_rse(row),
+                   error = function(e) { LOG("  create_rse failed: ", conditionMessage(e)); NULL })
+   if (is.null(rse)) return(NULL)
+   cd  <- as.data.frame(SummarizedExperiment::colData(rse))
+   stc <- intersect(c("tcga.cgc_sample_sample_type",
+                      "tcga.gdc_cases.samples.sample_type",
+                      "tcga.xml_sample_type"), colnames(cd))[1]
+   if (is.na(stc)) { LOG("  no sample-type column found."); return(NULL) }
+   tab <- sort(table(cd[[stc]]), decreasing = TRUE)
+   LOG("  field: ", stc)
+   for (i in seq_along(tab)) LOG("    ", names(tab)[i], " = ", tab[i])
+   data.frame(project = project, sample_type = names(tab), n = as.integer(tab), row.names = NULL)
+ }
